\blok{Kvadratická rovnice s parametrem}
Parametr ovlivňuje stupeň, diskriminant i počet kořenů; oddělená diskuse výjimečných hodnot.
\blok{Metoda a návaznost na graf}
\begin{itemize}
\item Zkoumej nejdříve, kdy se koeficient u $x^2$ rovná nule.
\item Pro zbývající hodnoty parametru rozhodne znaménko diskriminantu.
\end{itemize}
\blok{Společně řešené příklady}
\begin{prikladReseniBox}[title={Příklad 1}]
$(a-1)x^2-4=0$
\tcblower
\[
\begin{aligned}
a&=1:\quad -4=0\Rightarrow K_1=\varnothing\\[2pt]
a&>1:\quad x^2=\frac4{a-1}\\[2pt]
K_a&=\left\{-\frac2{\sqrt{a-1}};\frac2{\sqrt{a-1}}\right\}\\[2pt]
a&<1:\quad K_a=\varnothing
\end{aligned}
\]
\end{prikladReseniBox}
\begin{prikladReseniBox}[title={Příklad 2}]
$x^2-2ax+a^2-1=0$
\tcblower
\[
\begin{aligned}
(x-a)^2&=1\\[2pt]
x&=a-1\quad\text{nebo}\quad x=a+1\\[2pt]
K_a&=\{a-1;a+1\}
\end{aligned}
\]
\end{prikladReseniBox}
\blok{Samostatné procvičování}
\begin{zadaniReseniBox}[title={Úloha 1}]
$ax^2-2x+1=0$
\tcblower
\[
\begin{aligned}
a&=0:\quad -2x+1=0\Rightarrow K_0=\{\tfrac12\}\\[2pt]
a&\ne0:\quad D=4-4a\\[2pt]
a&>1:\quad K_a=\varnothing\\[2pt]
a&=1:\quad K_1=\{1\}\\[2pt]
a&<1,\ a\ne0:\quad K_a=\left\{\frac{1-\sqrt{1-a}}a;\frac{1+\sqrt{1-a}}a\right\}
\end{aligned}
\]
\end{zadaniReseniBox}
\begin{zadaniReseniBox}[title={Úloha 2}]
$(a+2)x^2-4(a+2)=0$
\tcblower
\[
\begin{aligned}
a&=-2:\quad 0=0\ \Rightarrow\ K_{-2}=\mathbb R\\[2pt]
a&\ne-2:\quad x^2=4\\[2pt]
K_a&=\{-2;2\}
\end{aligned}
\]
\end{zadaniReseniBox}
\blok{Volitelná rezerva}
\poznamka{Studenti si vymyslí obdobný příklad se změnou jednoho koeficientu a vysvětlí, jak se změní počet kořenů či řešení.}
\blok{Domácí úkol}
\begin{zadaniReseniBox}[title={Domácí příprava}]
\begin{enumerate}[label=\alph*),itemsep=2pt]
\item $ax^2-2x+1=0$
\item $(a+2)x^2-4(a+2)=0$
\item $x^2-2ax+a^2=0$
\item $(a-3)x^2=9(a-3)$
\item $(a-1)x^2-4=0$
\item $x^2-2ax+a^2-1=0$
\end{enumerate}
\tcblower
\begin{enumerate}[label=\alph*),itemsep=2pt]
\item
\[\begin{aligned}
a&=0:\quad -2x+1=0\Rightarrow K_0=\{\tfrac12\}\\[2pt]
a&\ne0:\quad D=4-4a\\[2pt]
a&>1:\quad K_a=\varnothing\\[2pt]
a&=1:\quad K_1=\{1\}\\[2pt]
a&<1,\ a\ne0:\quad x_{1,2}=\frac{1\pm\sqrt{1-a}}a
\end{aligned}\]
\item
\[\begin{aligned}
a&=-2:\quad 0=0\ \Rightarrow\ K_{-2}=\mathbb R\\[2pt]
a&\ne-2:\quad x^2=4\\[2pt]
K_a&=\{-2;2\}
\end{aligned}\]
\item
\[\begin{aligned}
(x-a)^2&=0\\[2pt]
x&=a\\[2pt]
K_a&=\{a\}
\end{aligned}\]
\item
\[\begin{aligned}
a&=3:\quad 0=0\ \Rightarrow\ K_3=\mathbb R\\[2pt]
a&\ne3:\quad x^2=9\\[2pt]
K_a&=\{-3;3\}
\end{aligned}\]
\item
\[\begin{aligned}
a&=1:\quad -4=0\Rightarrow K_1=\varnothing\\[2pt]
a&>1:\quad x^2=\frac4{a-1}\\[2pt]
K_a&=\left\{-\frac2{\sqrt{a-1}};\frac2{\sqrt{a-1}}\right\}\\[2pt]
a&<1:\quad K_a=\varnothing
\end{aligned}\]
\item
\[\begin{aligned}
(x-a)^2&=1\\[2pt]
x&=a-1\quad\text{nebo}\quad x=a+1\\[2pt]
K_a&=\{a-1;a+1\}
\end{aligned}\]
\end{enumerate}
\end{zadaniReseniBox}

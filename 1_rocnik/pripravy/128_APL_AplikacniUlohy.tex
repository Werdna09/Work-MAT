\blok{Kvadratické úlohy z praxe}
Reálné vztahy v mechanice, elektrotechnice a geometrii s výukovými daty.
\blok{Metoda a návaznost na graf}
\begin{itemize}
\item Vhodné kořeny vybírej podle reálných podmínek veličin; porovnávej algebraický a věcný výsledek.
\item Nepleť si diskriminant modelového vztahu s důkazem fyzikální platnosti modelu.
\end{itemize}
\blok{Společně řešené příklady}
\begin{prikladReseniBox}[title={Příklad 1}]
Model výšky míčku je $h(t)=-5t^2+20t+1$ (metry, sekundy). Kdy je míček ve výšce $16\,\mathrm m$?
\tcblower
\[
\begin{aligned}
-5t^2+20t+1&=16\\[2pt]
t^2-4t+3&=0\\[2pt]
(t-1)(t-3)&=0\\[2pt]
t&=1\,\mathrm s\ \text{nebo}\ 3\,\mathrm s
\end{aligned}
\]
\end{prikladReseniBox}
\begin{prikladReseniBox}[title={Příklad 2}]
Výnos výrobního procesu je modelován $Z(q)=-2q^2+40q-150$ (tis. Kč). Pro které nezáporné $q$ je $Z(q)>0$?
\tcblower
\[
\begin{aligned}
-2q^2+40q-150&>0\\[2pt]
q^2-20q+75&<0\\[2pt]
(q-5)(q-15)&<0\\[2pt]
K&=(5;15)
\end{aligned}
\]
\end{prikladReseniBox}
\begin{prikladReseniBox}[title={Příklad 3}]
Model výkonové ztráty odporu je $P=I^2R$, $R=8\,\Omega$. Pro jaký nezáporný proud dosahuje $P=72\,\mathrm W$?
\tcblower
\[
\begin{aligned}
72&=8I^2\\[2pt]
I^2&=9\\[2pt]
I&=3\,\mathrm A\quad(I\ge0)
\end{aligned}
\]
\end{prikladReseniBox}
\blok{Samostatné procvičování}
\begin{zadaniReseniBox}[title={Úloha 1}]
Dráha brzdění je v modelu $s=\frac{v^2}{2a}$, $a=5\,\mathrm{m/s^2}$. Jakou počáteční rychlost (v m/s) odpovídá dráha $40\,\mathrm m$?
\tcblower
\[
\begin{aligned}
40&=\frac{v^2}{10}\\[2pt]
v^2&=400\\[2pt]
v&=20\,\mathrm{m/s}\quad(v\ge0)
\end{aligned}
\]
\end{zadaniReseniBox}
\begin{zadaniReseniBox}[title={Úloha 2}]
Časový model letu dronu je $h(t)=-2t^2+12t$ metrů. Kdy je ve výšce $10\,\mathrm m$?
\tcblower
\[
\begin{aligned}
-2t^2+12t&=10\\[2pt]
t^2-6t+5&=0\\[2pt]
(t-1)(t-5)&=0\\[2pt]
t&\in\{1;5\}\,\mathrm s
\end{aligned}
\]
\end{zadaniReseniBox}

\blok{Volitelná výzva -- slovní úloha}
\begin{zadaniReseniBox}[title={Rozšiřující příklad}]
Obdélník má délku o $3$ cm větší než šířku a obsah $40$ cm$^2$. Určete rozměry.
\tcblower
\[\begin{aligned}
x(x+3)&=40\\x^2+3x-40&=0\\(x+8)(x-5)&=0\\x&=5\ \text{(kladný rozměr)}\\\text{Rozměry}&:\ 5\,\mathrm{cm},\ 8\,\mathrm{cm}
\end{aligned}\]
\end{zadaniReseniBox}
\blok{Domácí úkol}
\begin{zadaniReseniBox}[title={Domácí příprava}]
\begin{enumerate}[label=\alph*),itemsep=2pt]
\item Obdélník má obsah $60\,\mathrm{cm^2}$ a delší strana je o $7\,\mathrm{cm}$ větší. Určete délky stran.
\item Čtvercový pozemek má po rozšíření každé strany o $3\,\mathrm m$ obsah $100\,\mathrm{m^2}$. Jaká byla původní strana?
\item Dráha brzdění je v modelu $s=\frac{v^2}{2a}$, $a=5\,\mathrm{m/s^2}$. Jakou počáteční rychlost (v m/s) odpovídá dráha $40\,\mathrm m$?
\item Výnos výrobního procesu je modelován $Z(q)=-2q^2+40q-150$ (tis. Kč). Pro které nezáporné $q$ je $Z(q)>0$?

\item Časový model letu dronu je $h(t)=-2t^2+12t$ metrů. Kdy je ve výšce $10\,\mathrm m$?
\end{enumerate}
\tcblower
\begin{enumerate}[label=\alph*),itemsep=2pt]
\item
\[\begin{aligned}
x(x+7)&=60\\[2pt]
x^2+7x-60&=0\\[2pt]
(x-5)(x+12)&=0\\[2pt]
x&=5\,\mathrm{cm}\quad (x=-12\text{ nelze})\\[2pt]
\text{Strany: }&5\,\mathrm{cm};\ 12\,\mathrm{cm}
\end{aligned}\]
\item
\[\begin{aligned}
(x+3)^2&=100\\[2pt]
x+3&=10\quad (x>0)\\[2pt]
x&=7\,\mathrm m
\end{aligned}\]
\item
\[\begin{aligned}
40&=\frac{v^2}{10}\\[2pt]
v^2&=400\\[2pt]
v&=20\,\mathrm{m/s}\quad(v\ge0)
\end{aligned}\]
\item
\[\begin{aligned}
-2q^2+40q-150&>0\\[2pt]
q^2-20q+75&<0\\[2pt]
(q-5)(q-15)&<0\\[2pt]
K&=(5;15)
\end{aligned}\]
\item
\[\begin{aligned}
72&=8I^2\\[2pt]
I^2&=9\\[2pt]
I&=3\,\mathrm A\quad(I\ge0)
\end{aligned}\]
\item
\[\begin{aligned}
-2t^2+12t&=10\\[2pt]
t^2-6t+5&=0\\[2pt]
(t-1)(t-5)&=0\\[2pt]
t&\in\{1;5\}\,\mathrm s
\end{aligned}\]
\end{enumerate}
\end{zadaniReseniBox}

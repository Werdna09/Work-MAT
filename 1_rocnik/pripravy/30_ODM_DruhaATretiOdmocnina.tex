\blok{Výklad a souvislosti}

Ze známého obsahu čtverce hledáme délku jeho strany. Požadavek nezáporné délky vysvětluje, proč zápis $\sqrt9$ označuje pouze $3$. Třetí mocnina naproti tomu rozlišuje kladná a záporná čísla i bez omezení znaménka.

\begin{definiceBox}[title={Definice},breakable=false]
\textbf{Druhá odmocnina} z $a\ge0$ je jediné nezáporné číslo $b$, pro které platí $b^2=a$. Zapisujeme $b=\sqrt a$. Pro $a<0$ není druhá odmocnina v $\mathbb R$ definována.
\end{definiceBox}


\begin{definiceBox}[title={Definice},breakable=false]
\textbf{Třetí odmocnina} z reálného čísla $a$ je jediné reálné číslo $b$, pro které platí $b^3=a$. Zapisujeme $b=\sqrt[3]a$.
\end{definiceBox}


Každý výsledek ověř příslušným umocněním a u druhé odmocniny také znaménkem. Zatím nepoužíváme pravidla pro odmocninu součinu. Odhad získáme porovnáním se známými druhými nebo třetími mocninami.

\blok{Řešené příklady}

\begin{prikladReseniBox}[title={Příklad 1},breakable=false]
Urči $\sqrt{25}$, $\sqrt[3]{27}$, $\sqrt[3]{-8}$ a $\sqrt{-9}$.
\tcblower
Postupně $5$, $3$, $-2$; poslední výraz nemá v $\mathbb R$ smysl. Kontroly: $5^2=25$, $3^3=27$, $(-2)^3=-8$.
\end{prikladReseniBox}


\begin{prikladReseniBox}[title={Příklad 2},breakable=false]
Porovnej $\sqrt{16}$ s čísly, jejichž druhá mocnina je $16$. Odhadni $\sqrt{39}$.
\tcblower
$\sqrt{16}=4$. Druhou mocninu $16$ mají $4$ i $-4$, ale odmocnina je nezáporná. Protože $6^2<39<7^2$, platí $6<\sqrt{39}<7$.
\end{prikladReseniBox}


\blok{Samostatné procvičení}

\begin{zadaniReseniBox}[title={Procvičení},breakable=false]
\begin{enumerate}[label=\alph*)]
\item Vypočítej $\sqrt{0}$, $\sqrt{0{,}09}$ a $\sqrt[3]{0{,}001}$.
\item Vypočítej $\sqrt[3]{-125}$ a $\sqrt{(-7)^2}$.
\item Najdi dvě sousední celá čísla, mezi nimiž leží $\sqrt7$.
\end{enumerate}
\tcblower
\begin{enumerate}[label=\alph*)]
\item $0$, $0{,}3$ a $0{,}1$; ověříme umocněním.
\item $-5$ a $7$; druhá odmocnina vrací nezáporné číslo.
\item $2<\sqrt7<3$, protože $4<7<9$.
\end{enumerate}
\end{zadaniReseniBox}


\poznamka{Nepiš $\sqrt{16}=\pm4$. Nezaměňuj existenci přesné hodnoty odmocniny s možností vyjádřit ji konečným desetinným zápisem. Iracionální číslo má neperiodický desetinný rozvoj. Tuto skutečnost připomeň, ale nevyžaduj nový důkaz.}


\blok{Navazující procvičování ve třídě}

\begin{zadaniReseniBox}[title={Základ: hodnota odmocniny a kontrola},breakable=false]
\begin{enumerate}[label=\alph*)]
\item Urči $\sqrt{144}$, $\sqrt{0{,}0004}$, $\sqrt[3]{-64}$ a $\sqrt[3]{0{,}008}$. Každý výsledek ověř mocninou.
\item Porovnej $-\sqrt{49}$, $\sqrt{(-7)^2}$ a $\sqrt{-49}$ v reálných číslech.
\item Urči všechna reálná čísla, jejichž druhá mocnina je $36$, a porovnej s hodnotou $\sqrt{36}$.
\end{enumerate}
\tcblower
\begin{enumerate}[label=\alph*)]
\item $12$, $0{,}02$, $-4$, $0{,}2$. Kontroly: $12^2=144$, $0{,}02^2=0{,}0004$, $(-4)^3=-64$, $0{,}2^3=0{,}008$.
\item První hodnota je $-7$, druhá $7$, třetí v $\mathbb R$ neexistuje.
\item Čísla $-6$ i $6$, ale odmocnina má jedinou nezápornou hodnotu $6$.
\end{enumerate}
\end{zadaniReseniBox}

\begin{zadaniReseniBox}[title={Rezerva: odhad a vnořené odmocniny},breakable=false]
\begin{enumerate}[label=\alph*)]
\item Najdi dvě po sobě jdoucí celá čísla, mezi nimiž leží $\sqrt{70}$ a $\sqrt[3]{-30}$.
\item Vypočítej $\sqrt{\sqrt{81}}$ a $\sqrt[3]{\sqrt{64}}$.
\end{enumerate}
\tcblower
\begin{enumerate}[label=\alph*)]
\item $8<\sqrt{70}<9$, protože $64<70<81$; $-4<\sqrt[3]{-30}<-3$, protože $-64<-30<-27$.
\item $\sqrt9=3$ a $\sqrt[3]8=2$.
\end{enumerate}
\end{zadaniReseniBox}


\poznamka{Základ: původní definice a nová číselná skupina s ověřením umocněním. Odhady a vnořené odmocniny jsou rezerva. Domácí pětice upevňuje rozdíl mezi záporným radikandem a záporným výsledkem. Pravidla pro součin odmocnin a částečné odmocňování ponecháme hodinám 33 a 34.}

\blok{Domácí procvičení}

\begin{zadaniReseniBox}[title={Domácí úlohy},breakable=false]
\begin{enumerate}[label=\alph*)]
\item Vypočítej $\sqrt{225}$, $\sqrt{0{,}0025}$, $\sqrt[3]{-216}$.
\item Vypočítej $-\sqrt{(-5)^2}$ a $\sqrt[3]{(-5)^3}$.
\item Rozhodni, zda v $\mathbb R$ existuje $\sqrt{-0{,}01}$ a $\sqrt[3]{-0{,}001}$.
\item Odhadni $\sqrt{19}$ mezi po sobě jdoucími celými čísly.
\item Najdi chybu: $\sqrt{100}=\pm10$.
\end{enumerate}
\tcblower
\begin{enumerate}[label=\alph*)]
\item $15$, $0{,}05$, $-6$.
\item $-5$ a $-5$.
\item Druhá odmocnina neexistuje, třetí existuje a rovná se $-0{,}1$.
\item $4<\sqrt{19}<5$, neboť $16<19<25$.
\item Odmocnina je nezáporné číslo $10$. Obě hodnoty $\pm10$ řeší otázku, která čísla mají druhou mocninu $100$.
\end{enumerate}
\end{zadaniReseniBox}

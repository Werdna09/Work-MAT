\nastavPripravu{1. ročník}{Odmocniny}{35}{Usměrňování jednočlenného jmenovatele}

\begin{document}

\blok{Výklad a souvislosti}

Hodnotu zlomku nezměníme, násobíme-li čitatele i jmenovatele stejným nenulovým číslem. Zvolíme takové číslo, aby ve jmenovateli vznikla úplná mocnina, kterou umíme odmocnit.

\begin{definiceBox}[title={Definice},breakable=false]
\textbf{Usměrnění jmenovatele} je úprava zlomku, která odstraní odmocninu ze jmenovatele a zachová jeho hodnotu na původním definičním oboru.
\end{definiceBox}


Pro $b>0$ platí $\frac a{\sqrt b}=\frac{a\sqrt b}{b}$, protože rozšiřujeme zlomkem $\sqrt b/\sqrt b=1$. Nejdříve může být výhodné odmocninu částečně odmocnit a zlomek zkrátit. U třetí odmocniny doplňujeme třetí mocninu; například $\sqrt[3]2\,\sqrt[3]4=2$.

Volba činitele vychází z rozkladu radikandu. U $\sqrt[3]{18}$
je $18=2\cdot3^2$: do třetích mocnin chybějí dvě dvojky a jedna trojka.
Rozšíříme proto $\sqrt[3]{12}$, neboť
$\sqrt[3]{18}\sqrt[3]{12}=\sqrt[3]{2^3\cdot3^3}=6$.
Můžeme rozšiřovat i větším činitelem, ale pak bude více krácení.
Vhodný činitel tedy hledáme podle chybějících exponentů,
nemusíme se jej učit pro každý příklad zvlášť.

\blok{Řešené příklady}

\begin{prikladReseniBox}[title={Příklad 1},breakable=false]
Usměrni $\frac3{\sqrt5}$ a $\frac5{\sqrt{20}}$.
\tcblower
$\frac3{\sqrt5}=\frac{3\sqrt5}{5}$. Dále $\frac5{\sqrt{20}}=\frac5{2\sqrt5}=\frac{5\sqrt5}{10}=\frac{\sqrt5}{2}$. V obou případech je původní jmenovatel nenulový.
\end{prikladReseniBox}


\begin{prikladReseniBox}[title={Příklad 2},breakable=false]
Usměrni $\frac1{\sqrt x}$ a uveď podmínky.
\tcblower
Odmocnina i nenulový jmenovatel vyžadují $x>0$. Potom $\frac1{\sqrt x}\cdot\frac{\sqrt x}{\sqrt x}=\frac{\sqrt x}{x}$.
\end{prikladReseniBox}


\blok{Samostatné procvičení}

\begin{zadaniReseniBox}[title={Procvičení},breakable=false]
\begin{enumerate}[label=\alph*)]
\item Usměrni $\frac2{\sqrt3}$.
\item Usměrni $\frac6{\sqrt{12}}$.
\item Usměrni $\frac1{\sqrt[3]4}$.
\end{enumerate}
\tcblower
\begin{enumerate}[label=\alph*)]
\item $\frac{2\sqrt3}{3}$. Rozšiřujeme nenulovým $\sqrt3$.
\item $\frac6{2\sqrt3}=\frac3{\sqrt3}=\sqrt3$.
\item $\frac{\sqrt[3]2}{2}$, protože $\sqrt[3]4\,\sqrt[3]2=\sqrt[3]8=2$.
\end{enumerate}
\end{zadaniReseniBox}


\poznamka{Požaduj vysvětlení, proč zvolený rozšiřující zlomek představuje jedničku. Nestačí násobit pouze jmenovatel. Dvojčlenné jmenovatele a sdružené výrazy ponech 36. hodině; nepředváděj je zde ani jako rezervu.}


\blok{Navazující procvičování ve třídě}

\begin{zadaniReseniBox}[title={Základ: nejprve zjednoduš, potom usměrni},breakable=false]
\begin{enumerate}[label=\alph*)]
\item Usměrni $\frac4{\sqrt8}$ a $\frac{3\sqrt2}{2\sqrt3}$.
\item Usměrni $\frac1{\sqrt[3]2}$ a $\frac2{\sqrt[3]9}$. Jakou třetí mocninu vytváříš ve jmenovateli?
\item Urči podmínky a usměrni $\frac2{3\sqrt x}$.
\end{enumerate}
\tcblower
\begin{enumerate}[label=\alph*)]
\item $\frac4{2\sqrt2}=\sqrt2$; $\frac{3\sqrt6}{6}=\frac{\sqrt6}2$.
\item $\frac{\sqrt[3]4}{2}$ a $\frac{2\sqrt[3]3}{3}$. Jmenovatele po rozšíření jsou $\sqrt[3]8=2$ a $\sqrt[3]{27}=3$.
\item $x>0$ a $\frac{2\sqrt x}{3x}$.
\end{enumerate}
\end{zadaniReseniBox}

\begin{zadaniReseniBox}[title={Rezerva: znaménko a proměnná},breakable=false]
\begin{enumerate}[label=\alph*)]
\item Usměrni $\frac1{\sqrt{x^2}}$ pro reálné $x$.
\item Usměrni $\frac3{\sqrt[3]{-4}}$.
\end{enumerate}
\tcblower
\begin{enumerate}[label=\alph*)]
\item Podmínka $x\ne0$. Již $\sqrt{x^2}=|x|$ dává $1/|x|$ bez odmocniny ve jmenovateli; lze také psát $|x|/x^2$.
\item Rozšíříme $\sqrt[3]2$: $\frac{3\sqrt[3]2}{\sqrt[3]{-8}}=-\frac{3\sqrt[3]2}{2}$.
\end{enumerate}
\end{zadaniReseniBox}


\poznamka{Základ: původní ukázky, nová první skupina a samostatné procvičení. Druhá skupina je rezerva pro rychlé. Domácí úlohy mají procvičit i volbu rozšiřujícího činitele u třetí odmocniny; dvě nepřeskočitelné otázky jsou existence odmocniny a nenulovost jmenovatele. Součet ve jmenovateli necháme další hodině.}

\blok{Domácí procvičení}

\begin{zadaniReseniBox}[title={Domácí úlohy},breakable=false]
\begin{enumerate}[label=\alph*)]
\item Usměrni $\frac7{\sqrt7}$ a $\frac5{2\sqrt5}$.
\item Usměrni $\frac3{\sqrt{27}}$.
\item Usměrni $\frac1{\sqrt[3]5}$.
\item Urči podmínky a usměrni $\frac4{\sqrt a}$.
\item Najdi chybu: $\frac1{\sqrt3}=\frac{\sqrt3}{\sqrt3}$.
\end{enumerate}
\tcblower
\begin{enumerate}[label=\alph*)]
\item $\sqrt7$ a $\frac{\sqrt5}2$.
\item $1/\sqrt3=\sqrt3/3$.
\item $\frac{\sqrt[3]{25}}5$.
\item $a>0$, výsledek $\frac{4\sqrt a}{a}$.
\item Ve jmenovateli musí být součin $\sqrt3\sqrt3=3$. Správně $\sqrt3/3$, neboť rozšiřujeme celý zlomek.
\end{enumerate}
\end{zadaniReseniBox}

\blok{Připomenutí: součin jednočlenů}
Z 41. hodiny víme, že jednočlen je součin koeficientu a mocnin proměnných. Při násobení jednočlenů vynásobíme koeficienty a exponenty stejných proměnných \emph{sečteme}.
\begin{definiceBox}[title={Pravidlo: součin jednočlenů}]
Pro $a,b\in\mathbb R$ a $m,n\in\mathbb N_0$ platí
\[(ax^m)(bx^n)=abx^{m+n}.\]
Stejné pravidlo se použije zvlášť pro každou další proměnnou.
\end{definiceBox}
\begin{prikladReseniBox}[title={Rozcvička: dva jednočleny}]
Vynásob $(-2x^2y)(3xy^3)$.
\tcblower
$(-2)\cdot3\cdot x^{2+1}y^{1+3}=-6x^3y^4$.
\end{prikladReseniBox}
\blok{Násobení jednočlenu a mnohočlenu}
Z čísel známe $3(4+5)=3\cdot4+3\cdot5$. Stejný \textbf{distributivní zákon} platí i pro algebraické výrazy.
\begin{definiceBox}[title={Pravidlo: jednočlen krát mnohočlen}]
Pro jednočlen $M$ a mnohočlen $P=A_1+\dots+A_k$ platí
\[M\cdot P=M A_1+\dots+M A_k.\]
Každý člen mnohočlenu násobíme \textbf{celým} jednočlenem včetně znaménka.
\end{definiceBox}
\begin{prikladReseniBox}[title={Příklad 1: kladný jednočlen}]
Vynásob $3x(2x^2-4x+5)$.
\tcblower
\begin{align*}
3x(2x^2-4x+5)&=3x\cdot2x^2+3x\cdot(-4x)+3x\cdot5\\
&=6x^3-12x^2+15x.
\end{align*}
\end{prikladReseniBox}
\begin{prikladReseniBox}[title={Příklad 2: záporný jednočlen}]
Vynásob $-2a^2(3a^3-4a+7)$.
\tcblower
$-6a^5+8a^3-14a^2$.
\end{prikladReseniBox}
\blok{Násobení dvou mnohočlenů}
Nejprve rozepíšeme první závorku po členech; \emph{každým} z nich vynásobíme celou druhou závorku. Nakonec sloučíme stejnorodé členy (42. hodina).
\begin{definiceBox}[title={Pravidlo: mnohočlen krát mnohočlen}]
Na součinu dvou dvojčlenů je vidět dvojí použití distributivity:
\[(a+b)(c+d)=ac+ad+bc+bd.\]
Obecně \textbf{každý člen první závorky násobíme každým členem druhé}.
\end{definiceBox}
\begin{prikladReseniBox}[title={Příklad 3: dvojčlen krát dvojčlen}]
Vynásob $(2x+3)(x-4)$.
\tcblower
\begin{align*}
(2x+3)(x-4)&=2x\cdot x+2x\cdot(-4)+3\cdot x+3\cdot(-4)\\
&=2x^2-8x+3x-12=2x^2-5x-12.
\end{align*}
\end{prikladReseniBox}
\begin{prikladReseniBox}[title={Příklad 4: trojčlen krát dvojčlen (podle učebnice)}]
Vynásob $(5x^3+6x-1)(x+2)$.
\tcblower
\begin{align*}
&5x^3(x+2)+6x(x+2)-1(x+2)\\
&=5x^4+10x^3+6x^2+12x-x-2\\
&=5x^4+10x^3+6x^2+11x-2.
\end{align*}
\end{prikladReseniBox}
\poznamka{Pro součin dvojčlenu a trojčlenu očekáváme před slučováním šest dílčích součinů. Upozorni na člen $-1$, kterým násobíme oba členy druhé závorky. Algebraické vzorce $ (a+b)^2$ apod. nepoužívej jako hotová pravidla; patří až do 44. hodiny.}
\blok{Samostatné procvičení}
\begin{zadaniReseniBox}[title={Základ: jednočlen a dvě závorky}]
\begin{enumerate}[label=\alph*)]
\item $4x(x^2-3x+2)$.
\item $-5a(2a^2-a-1)$.
\item $(x+1)(x-4)$.
\item $(3x-2)(2x+5)$.
\item $(2x+y)(x-2y)$.
\end{enumerate}
\tcblower
\begin{enumerate}[label=\alph*)]
\item $4x^3-12x^2+8x$.
\item $-10a^3+5a^2+5a$.
\item $x^2-3x-4$.
\item $6x^2+11x-10$.
\item $2x^2-3xy-2y^2$.
\end{enumerate}
\end{zadaniReseniBox}
\begin{zadaniReseniBox}[title={Návaznost: násobení a odčítání}]
Uprav $(x+2)(x-3)-(x-1)(x+4)$.
\tcblower
$(x^2-x-6)-(x^2+3x-4)=-4x-2$.
\end{zadaniReseniBox}
\blok{Volitelná rezerva}
\begin{zadaniReseniBox}[title={Trojčlen krát trojčlen}]
Roznásob $(x^2+x+1)(x^2-x+1)$.
\tcblower
Po vynásobení všech devíti dvojic členů a sloučení: $x^4+x^2+1$.
\end{zadaniReseniBox}
\blok{Domácí procvičení}
\begin{zadaniReseniBox}[title={Domácí úlohy}]
\begin{enumerate}[label=\alph*)]
\item $7x(2x^2-x+3)$.
\item $(2x+1)(x-3)$.
\item $(a-2)(a^2-3a+4)$.
\item $(x+2y)(3x-y)$.
\item $-2ab(3a^2-2ab+5b^2)$.
\end{enumerate}
\tcblower
\begin{enumerate}[label=\alph*)]
\item $14x^3-7x^2+21x$.
\item $2x^2-5x-3$.
\item $a^3-5a^2+10a-8$.
\item $3x^2+5xy-2y^2$.
\item $-6a^3b+4a^2b^2-10ab^3$.
\end{enumerate}
\end{zadaniReseniBox}

\begin{document}

\blok{Výklad a souvislosti}

Dělení součtu nenulovým jednočlenem rozložíme na podíly jednotlivých členů. Na rozdíl od násobení však musíme od začátku vyloučit všechny hodnoty, pro které se dělitel rovná nule.

\begin{definiceBox}[title={Připomenutí definice},breakable=false]
\textbf{Podíl} $A:M$ je výraz $A/M$ definovaný jen pro $M\ne0$. Přípustnost dělitele posuzujeme před krácením.
\end{definiceBox}


Platí $\frac{A+B-C}{M}=\frac A M+\frac B M-\frac C M$ pro $M\ne0$. V každém podílu vyděl koeficienty a použij pravidlo pro mocniny. Výsledek nemusí být mnohočlenem, jestliže v některém členu zůstane proměnná ve jmenovateli.

\blok{Řešené příklady}

\begin{prikladReseniBox}[title={Příklad 1},breakable=false]
Vyděl $(4x^6+8x^3+2x^2):(2x^2)$.
\tcblower
Pro $x\ne0$ dostaneme $2x^4+4x+1$. Kontrola násobením: $2x^2(2x^4+4x+1)=4x^6+8x^3+2x^2$.
\end{prikladReseniBox}


\begin{prikladReseniBox}[title={Příklad 2},breakable=false]
Uprav $(x^2+1):x$.
\tcblower
Při $x\ne0$ je podíl $x+1/x$. Druhý člen nelze vynechat; podíl tedy není mnohočlen.
\end{prikladReseniBox}


\blok{Samostatné procvičení}

\begin{zadaniReseniBox}[title={Procvičení},breakable=false]
\begin{enumerate}[label=\alph*)]
\item Uprav $(6a^3-9a^2+3a):(3a)$.
\item Uprav $(8x^2y-4xy^2):(4xy)$.
\item Uprav $(2x^2-3):x^2$.
\end{enumerate}
\tcblower
\begin{enumerate}[label=\alph*)]
\item $2a^2-3a+1$ při $a\ne0$.
\item $2x-y$ při $x,y\ne0$; podmínky pocházejí z celého dělitele.
\item $2-3/x^2$ při $x\ne0$.
\end{enumerate}
\end{zadaniReseniBox}


\poznamka{Nezaměňuj dělení součtu za krácení vybraného sčítance. Každý člen musí být vydělen. Zkoušku násobením prováděj s celým získaným podílem. Dlouhé dělení mnohočlennou závorkou začíná až v další hodině.}


\blok{Navazující procvičování ve třídě}

\begin{zadaniReseniBox}[title={Základ: rozděl celý čitatel},breakable=false]
\begin{enumerate}[label=\alph*)]
\item Uprav $(12a^4b^2-8a^3b+4a^2b):(4a^2b)$.
\item Uprav $(6x^3-9x+3):(-3x)$.
\item Uprav $(\frac32x^3-\frac34x^2):(\frac34x^2)$.
\end{enumerate}
\tcblower
\begin{enumerate}[label=\alph*)]
\item $3a^2b-2a+1$ při $a,b\ne0$.
\item $-2x^2+3-1/x$ při $x\ne0$. Výsledek není mnohočlen.
\item $2x-1$ při $x\ne0$.
\end{enumerate}
\end{zadaniReseniBox}

\begin{zadaniReseniBox}[title={Rezerva: dvě proměnné a zlomek ve výsledku},breakable=false]
\begin{enumerate}[label=\alph*)]
\item Uprav $(4x^2y-6xy^2+2y):(2xy)$.
\item Je správně $(x^2+x):x=x^2+1$? Oprav a ověř násobením.
\end{enumerate}
\tcblower
\begin{enumerate}[label=\alph*)]
\item $2x-3y+1/x$ při $x,y\ne0$.
\item Ne, správně $x+1$ při $x\ne0$, neboť $x(x+1)=x^2+x$.
\end{enumerate}
\end{zadaniReseniBox}


\poznamka{Základ: původní příklady a první nová skupina, včetně případu, kdy zůstane zlomek. Druhá skupina je rezerva. Domácí čtveřici zakonči zkouškou násobením celého podílu dělitelem. Podmínky zapisujeme před krácením; mnohočlenný dělitel ještě nerozdělujeme po členech.}

\blok{Domácí procvičení}

\begin{zadaniReseniBox}[title={Domácí úlohy},breakable=false]
\begin{enumerate}[label=\alph*)]
\item Uprav $(15x^4-10x^2+5x):(5x)$.
\item Uprav $(6a^2b-9ab^2):(-3ab)$.
\item Uprav $(2x^3-4x+6):(2x)$.
\item Uprav $(\frac12y^3+\frac34y^2):(\frac14y^2)$.
\end{enumerate}
\tcblower
\begin{enumerate}[label=\alph*)]
\item $3x^3-2x+1$, $x\ne0$.
\item $-2a+3b$, $a,b\ne0$.
\item $x^2-2+3/x$, $x\ne0$.
\item $2y+3$, $y\ne0$.
\end{enumerate}
\end{zadaniReseniBox}

\blok{Společný jmenovatel: stejný princip jako u zlomků}
Připomeň $\frac12+\frac13=\frac36+\frac26$. U proměnných se mění pouze to, že jmenovatel musíme \emph{rozložit a zvolit vhodný společný násobek}. Podmínky píšeme před úpravami.
\begin{zadaniBox}[title={Pravidlo pro společný jmenovatel}]
Pro $B(x)\ne0$ a $D(x)\ne0$ platí
\[\frac{A}{B}\pm\frac{C}{D}=\frac{AD\pm BC}{BD}.\]
Součin $BD$ funguje vždy, ale často je výhodnější nejmenší společný násobek jmenovatelů.
\end{zadaniBox}
\begin{prikladReseniBox}[breakable=false,title={Příklad 1: dva různé jmenovatele}]
Uprav $\displaystyle\frac1x+\frac1{x+1}$.
\tcblower
Pro $x\ne0,-1$ platí
\[\frac{x+1+x}{x(x+1)}=\frac{2x+1}{x(x+1)}.\]
\end{prikladReseniBox}
\begin{prikladReseniBox}[breakable=false,title={Příklad 2: nejdříve rozlož jmenovatele}]
Uprav $\displaystyle\frac1{x-2}+\frac2{x^2-4}$.
\tcblower
$x^2-4=(x-2)(x+2)$, proto $x\ne-2,2$. Po rozšíření
\[\frac{x+2}{(x-2)(x+2)}+\frac2{(x-2)(x+2)}=\frac{x+4}{x^2-4}.\]
\end{prikladReseniBox}
\begin{prikladReseniBox}[breakable=false,title={Příklad 3: opačné jmenovatele}]
Uprav $\displaystyle\frac2{a-2}-\frac1{2-a}$.
\tcblower
Protože $2-a=-(a-2)$, je $\frac2{a-2}+\frac1{a-2}=\frac3{a-2}$, $a\ne2$.
\end{prikladReseniBox}
\poznamka{Při odčítání vícečlenného čitatele vždy napiš závorku: $A-(B+C)=A-B-C$. Žáci si mnohdy dobře rozšíří jmenovatele, ale chybují právě ve znaménkách čitatele.}
\blok{Samostatná práce}
\begin{zadaniReseniBox}[breakable=false,title={Procvičení: od společného jmenovatele ke krácení}]
\begin{enumerate}[label=\alph*)]
\item $\frac3x-\frac1{2x}$;
\item $\frac{x}{x-1}-\frac{x+1}{x-1}$;
\item $\frac1{x-2}-\frac1{x+2}$;
\item $\frac{x+2}{x-1}-\frac{2x-1}{x-1}$;
\item $\frac{x}{x-1}-\frac1{x(x-1)}$.
\end{enumerate}
\tcblower
\begin{enumerate}[label=\alph*)]
\item $\frac5{2x}$, $x\ne0$.
\item $-\frac1{x-1}$, $x\ne1$.
\item $\frac4{x^2-4}$, $x\ne-2,2$.
\item $\frac{3-x}{x-1}$, $x\ne1$.
\item $\frac{x^2-1}{x(x-1)}=\frac{x+1}{x}$, $x\ne0,1$.
\end{enumerate}
\end{zadaniReseniBox}
\blok{Volitelná rezerva}
\begin{zadaniReseniBox}[breakable=false,title={Rozšíření: tři zlomky a tři podmínky}]
Uprav $\displaystyle\frac1{x-1}+\frac1{x+1}-\frac{2x}{x^2-1}$.
\tcblower
Pro $x\ne\pm1$ je čitatel $(x+1)+(x-1)-2x=0$, tedy celý výraz je $0$ jen na původním oboru.
\end{zadaniReseniBox}
\blok{Domácí procvičení}
\begin{zadaniReseniBox}[breakable=false,title={Domácí úkol: sčítání a odčítání}]
\begin{enumerate}[label=\alph*)]
\item $\frac2x+\frac3{2x}$;
\item $\frac1{x-3}+\frac1{x+3}$;
\item $\frac2{a-1}+\frac3{1-a}$;
\item $\frac{x}{x+2}-\frac1{x+2}$;
\item $\frac{x}{x+1}+\frac1{1-x}$;
\item $\frac3{x^2-1}-\frac1{x-1}$;
\item $\frac1x+\frac1{x+1}-\frac1{x(x+1)}$.
\end{enumerate}
\tcblower
\begin{enumerate}[label=\alph*)]
\item $\frac7{2x}$, $x\ne0$.
\item $\frac{2x}{x^2-9}$, $x\ne\pm3$.
\item $-\frac1{a-1}$, $a\ne1$.
\item $\frac{x-1}{x+2}$, $x\ne-2$.
\item $\frac{x^2-2x-1}{x^2-1}$, $x\ne\pm1$.
\item $\frac{2-x}{x^2-1}$, $x\ne\pm1$.
\item $\frac{2x}{x(x+1)}=\frac2{x+1}$, $x\ne0,-1$.
\end{enumerate}
\end{zadaniReseniBox}

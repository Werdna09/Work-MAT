\begin{center}\Large\bfseries 95. hodina -- Souhrnné procvičování\end{center}
\noindent\textbf{Jméno:} \rule{65mm}{0.4pt}\hfill\textbf{Třída:} \rule{24mm}{0.4pt}\par
\small Vyberte si úlohy podle potřeby. Doporučené minimum: jednu rovnici, jednu nerovnici, jednu soustavu a jednu úlohu s absolutní hodnotou. Zadání C jsou dobrovolná výzva. U výsledků napište množinu $K$ a všude zdůvodněte případné podmínky.\normalsize
\section*{A -- Rozcvičení}
\noindent\textbf{Úloha 1.} Vyřešte $4(x-2)=2x+6$.\par\vspace{6mm}
\noindent\textbf{Úloha 2.} Vyřešte $\frac{2x-1}{3}-\frac{x+2}{2}\le1$.\par\vspace{6mm}
\noindent\textbf{Úloha 3.} Vyřešte $\begin{cases}x+y=7,\\2x-y=2.\end{cases}$\par\vspace{6mm}
\noindent\textbf{Úloha 4.} Vyřešte $|x-3|=4$.\par\vspace{6mm}
\noindent\textbf{Úloha 5.} Vyřešte $|x-2|\le3$.\par\vspace{6mm}
\section*{B -- Samostatná práce}
\noindent\textbf{Úloha 6.} Vyřešte $3[2x-(x-4)]=2x+16$.\par\vspace{6mm}
\noindent\textbf{Úloha 7.} Vyřešte $2-3(x-1)>x+5$.\par\vspace{6mm}
\noindent\textbf{Úloha 8.} Vyřešte $\begin{cases}2x-3y=7,\\4x+y=9.\end{cases}$\par\vspace{6mm}
\noindent\textbf{Úloha 9.} Vyřešte $|x-2|+|x+2|=6$.\par\vspace{6mm}
\noindent\textbf{Úloha 10.} Diskutujte $(a-2)x=a+4$ pro všechna $a\in\mathbb R$.\par\vspace{6mm}
\noindent\textbf{Úloha 11.} Ze vzorce $Q=cm(T_2-T_1)$ vyjádřete $T_2$ včetně podmínky.\par\vspace{6mm}
\section*{C -- Výzva}
\noindent\textbf{Úloha 12.} Vyřešte $\begin{cases}x+y+z=6,\\2x-y+z=3,\\x+2y-z=2.\end{cases}$\par\vspace{6mm}
\noindent\textbf{Úloha 13.} Diskutujte $(a-1)x=a^2-1$ pro všechna $a\in\mathbb R$.\par\vspace{6mm}
\noindent\textbf{Úloha 14.} Vyřešte $\frac{x-1}{x+2}\ge0$.\par\vspace{6mm}
\noindent\textbf{Úloha 15.} Tarif A stojí $100+4m$ Kč, B stojí $220+2m$ Kč. Při kolika minutách stojí stejně? Který je levnější pro menší $m$?\par\vspace{6mm}
\noindent\textbf{Úloha 16.} Vyřešte $|x-2|+|x+2|=4$.\par\vspace{6mm}

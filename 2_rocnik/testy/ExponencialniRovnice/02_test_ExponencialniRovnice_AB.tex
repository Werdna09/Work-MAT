    Vyřešte následující exponenciální rovnice.
    \begin{enumerate}
        \item $9^x = 3$
        \AnswerSpace{18mm}
        \item $2^{3x-1} \cdot 4 = 8^{x+1} \cdot \left(\dfrac{1}{2}\right)^{x}$
        \AnswerSpace{24mm}
        \item Řešte pomocí vytýkání: $3^{x} \cdot \left(\dfrac{1}{2}\right)^{x} + 3^{x+1} \cdot \left(\dfrac{1}{2}\right)^{x+1} = \dfrac{5}{3}$
        \AnswerSpace{30mm}
        \item Řešte pomocí substituce: $4^{2x} - 6 \cdot 4^{x} + 8 = 0$
        \AnswerSpace{34mm}
    \end{enumerate}

    Vyřešte následující exponenciální rovnice.
    \begin{enumerate}
        \item $8^{x} = 2$
        \AnswerSpace{18mm}
        \item $\sqrt[4]{4^x} \cdot \sqrt[3]{2^{x-3}} = \sqrt[6]{16}$
        \AnswerSpace{24mm}
        \item Řešte pomocí vytýkání: $2^{2x} \cdot 5^{x} - 2^{2x-1} \cdot 5^{x+1} = -600$
        \AnswerSpace{30mm}
        \item Řešte pomocí substituce: $3^{2x} - 4 \cdot 3^{x} = -3$
        \AnswerSpace{34mm}
    \end{enumerate}

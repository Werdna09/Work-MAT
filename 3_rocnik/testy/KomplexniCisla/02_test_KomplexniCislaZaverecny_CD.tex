\TestHeader{C}

% ZADÁNÍ VARIANTY C
\begin{enumerate}
    \item Vypočítejte a výsledek zapište v algebraickém tvaru: $(2+i)(3+i)(1+2i)$
    \AnswerSpace{16mm}
    \item Vypočítejte $\displaystyle \frac{|(2+i)^2|}{|1-2i|}$.
    \AnswerSpace{22mm}
    \item Řešte v $\mathbb{C}$: $5x^2-2x+1=0$.
    \AnswerSpace{25mm}
    \item Jsou dána komplexní čísla $\displaystyle z_1=1-\sqrt3i$,
        $\displaystyle z_2=4\left(\cos\frac{\pi}{6}+i\sin\frac{\pi}{6}\right)$.
    Vypočítejte součin $z_1z_2$ a podíl $\frac{z_1}{z_2}$; oba výsledky zapište v goniometrickém i algebraickém tvaru.
    \AnswerSpace{37mm}
    \item Je dáno $\displaystyle
            z=\frac12\left(
                \cos\frac{4\pi}{9}
                +i\sin\frac{4\pi}{9}
            \right)$.
    Pomocí Moivreovy věty vypočítejte $\frac1{z^3}$ a výsledek zapište v algebraickém tvaru.
    \AnswerSpace{29mm}
\end{enumerate}

\TestHeader{D}

% ZADÁNÍ VARIANTY D
\begin{enumerate}
    \item Vypočítejte a výsledek zapište v algebraickém tvaru: $(1-2i)^3$
    \AnswerSpace{16mm}
    \item Vypočítejte $\displaystyle \left|\frac{(1-i)^3}{2+i}\right|$.
    \AnswerSpace{22mm}
    \item Řešte v $\mathbb{C}$: $3x^2-2x+1=0$.
    \AnswerSpace{25mm}
    \item Jsou dána komplexní čísla $\displaystyle z_1=-\frac{5\sqrt3}{2}-\frac52i$,
        $\displaystyle z_2=\cos\frac{\pi}{3}+i\sin\frac{\pi}{3}$.
    Vypočítejte součin $z_1z_2$ a podíl $\frac{z_1}{z_2}$; oba výsledky zapište v goniometrickém i algebraickém tvaru.
    \AnswerSpace{37mm}
    \item Je dáno $\displaystyle
            z=\frac13\left(
                \cos\frac{5\pi}{6}
                +i\sin\frac{5\pi}{6}
            \right)$.
    Pomocí Moivreovy věty vypočítejte $\frac1{z^2}$ a výsledek zapište v algebraickém tvaru.
    \AnswerSpace{29mm}
\end{enumerate}

\TestHeader{E}

% ZADÁNÍ VARIANTY E
\begin{enumerate}
    \item Upravte a výsledek zapište v algebraickém tvaru: $\displaystyle\left(\frac{2+i}{1-i}\right)^2$.
    \AnswerSpace{17mm}
    \item Vypočítejte $\displaystyle\left|\frac{\frac{i}{2-i}+\frac1i}{1+\frac{i}{2i+1}}\right|$.
    \AnswerSpace{28mm}
    \item Řešte v $\mathbb C$: $x^2-2\sqrt2x+5=0$.
    \AnswerSpace{27mm}
    \item Pomocí Moivreovy věty vypočítejte $(\sqrt3+i)^{10}$ a výsledek zapište v algebraickém tvaru.
    \AnswerSpace{30mm}
    \item Je dáno $\displaystyle z=3\left(\cos\frac{7\pi}{6}+i\sin\frac{7\pi}{6}\right)$.
    Pomocí Moivreovy věty vypočítejte $\frac1{z^2}$ a výsledek zapište v algebraickém tvaru.
    \AnswerSpace{30mm}
\end{enumerate}
